# 2. the REAL method (x0 metric, generate.py) at 1024, seeds 0 1, all 8 prompts.

python generate.py --name flux-dev --size 1024 --mode e2_full --tau 0.90 --merge-tmin 0.15 --smooth-sigma 1.0 --seeds 0 1 --text-cache text_cache.pt --out results/ours_x0_t090_s10
